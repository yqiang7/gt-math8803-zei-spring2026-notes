\chapter{Mar.~9 --- Affine Lie Algebras, Part 2}

\section{Integrable Representations, Continued}

\begin{remark}
  Recall that the Weyl module is
  given by
  \[
    V_{\lambda, k}
    = \Ind_{\widetilde{\g}^+}^{\widetilde{\g}} L_\lambda,
  \]
  where $L_\lambda$ is an irreducible
  representation for $\g$ and
  $\lambda \in P^+$. When
  $k \notin \Q \langle \alpha^\vee, \lambda \rangle + \Q$ for any $\alpha \in R_+$,
  we have that $V_{\lambda, k}$ is
  irreducible.
\end{remark}

\begin{definition}
  A highest weight representation
  $V$ of $\widetilde{\g}$ is \emph{integrable}
  if for all $u \in V$ and $0 \le i \le r$,
  \[
    f_i^{n_i + 1} u = 0
  \]
  for some positive integer $n_i$.
\end{definition}

\begin{remark}
  The action of $\widetilde{\g}$ on
  integrable representations can be extended
  to the corresponding centrally extended
  loop group.
\end{remark}

\begin{remark}
  Integrable representations are the analogues
  of finite-dimensional representations
  for $\widetilde{\g}$.
\end{remark}

\begin{theorem}
  We have the following:
  \begin{enumerate}
    \item The module $L_\Lambda$ is
      integrable if and only if
      $\Lambda \in \widehat{P}^+$.
    \item The representations $L_\Lambda$ for
      $\Lambda \in \widehat{P}^+$ are pairwise
      nonisomorphic, and any highest weight
      integrable module is isomorphic to
      one of the $L_\Lambda$.
    \item If $\Lambda$ is a dominant integral
      weight and $n_i = \langle \Lambda, \alpha_i^\vee \rangle$,
      then the $\widehat{\g}$-module $L_\Lambda$
      is generated by a highest weight
      vector $v_\Lambda$, with relations
      (for $0 \le i \le r$)
      \[
        e_i v_\Lambda = 0, \quad
        h_i v_\Lambda = n_i v_\Lambda, \quad
        f_i^{n_i + 1} v_\Lambda = 0.
      \]
  \end{enumerate}
\end{theorem}

\begin{remark}
  Let $V$ be a module for $\g$. For a fixed
  $z \in \C^\times$, define
  \[
    (x \otimes P(t)) u = P(z) x u
  \]
  for $u \in V$. If $V$ is finite-dimensional,
  then this representation $V(z)$ is
  an integrable $\widehat{\g}$-module (called
  an \emph{evaluation module}).
  Note that $V(z) = V[z, z^{-1}]$.
  Let $d = z (d / dz)$. Then we get a
  $\widetilde{\g}$-action on
  \[z^{-\Delta} V[z, z^{-1}] = V \otimes z^{-\Delta} \C[z, z^{-1}],\]
  where $d$ acts by $-\Delta + n$.
\end{remark}

\section{The Virasoro Algebra}\label{sec:virasoro}

\begin{remark}
  The \emph{Witt algebra} is generated
  by $L_n$ (for $n \in \Z$) with relations
  \[
    [L_m, L_n]
    = (m - n) L_{m + n}.
  \]
  If we write $L_n = -z^{n + 1} (d / dz)$
  for $z = e^{it}$, then the Witt algebra
  can be realized as
  vector fields on a circle. Consider
  $z \mapsto z + t z^{n + 1}$ and
  $f(z) \mapsto f(z + t z^{n + 1})$.
  The \emph{Virasoro algebra}
  $\mathrm{Vir}$ is generated
  by the $L_n$ and an additional
  element $K$, with relations
  \[
    [L_m, L_n]
    = (m - n) L_{m + n}
    + \frac{m^3 - m}{12} \delta_{m, -n} K,
    \quad [K, L_n] = 0.
  \]
  The Witt algebra and the Virasoro algebra
  act on $\widehat{\g}$ as follows:
  \[
    [L_m, x[n]]
    = -nx[m + n] \quad \text{and} \quad
    [L_n, c] = [K, \cdot] = 0.
  \]
  Note that the action of $L_0$ coincides with
  the action of $-d$.
  Thus we have a semidirect product
  $\mathrm{Vir} \ltimes \widehat{\g}$,
  which contains $\widetilde{\g}$ as a
  subalgebra.

  Let $V$ be a highest weight representation
  of $\widehat{\g}$ at level $k$, with
  $k \ne -h^\vee$. Define
  \[
    L_m = \frac{1}{2(k + h^\vee)}
    \sum_{a \in B} \sum_{n \in \Z}
    {:} a[n] a[m - n] {:}
    \quad \text{and} \quad
    K = \frac{k \dim \g}{k + h^\vee} \id, \tag{$*$}
  \]
  where $B$ is an orthonormal basis for $\g$.
  Here ${:} a[n] a[k] {:}$ is the
  \emph{normal ordered product}
  \[
    {:} a[n] a[k] {:}
    =
    \begin{cases}
      a[n] a[k] & \text{if } k \ge n, \\
      a[k] a[n] & \text{if } k < n,
    \end{cases}
  \]
  i.e. modes which kill the highest
  weight vector are pushed all the way to the
  right.
\end{remark}

\begin{theorem}
  The formulas $(*)$ define an action of
  the Virasoro algebra on $V$. This action
  with the original action of $\widehat{\g}$
  combine into a
  $(\mathrm{Vir} \ltimes \widehat{\g})$-action on $V$.
\end{theorem}

\begin{remark}
  Let $v$ be the highest weight vector. Then
  \[
    L_0 v
    = \frac{1}{2(k + h^\vee)}
    \sum_{n \in \Z} \sum_{a \in B} {:} a[n] a[-n] {:} v
    = \frac{1}{2(k + h^\vee)}
    \sum_{a \in B} {:} a[0] a[0] {:} v
    = \frac{(\lambda, \lambda + 2\rho)}{2(k + h^\vee)} v,
  \]
  where
  $\sum_{a \in B} {:} a[0] a[0] {:} = \sum_{a \in B} a^2$
  is the Casimir operator. Thus
  we can choose $\Delta$ so that
  $d = -L_0$.
\end{remark}

\section{Currents}

\begin{definition}
  Let $x[n] = x \otimes t^n$ for $x \in \g$.
  The \emph{current} associated to $x$ is
  \[
    J_x(z) = \sum_{n \in \Z} x[n] z^{-n - 1}
    = J_x^+(z) - J_x^-(z),
  \]
  where $J_x^+(z) = \sum_{n < 0} x[n] z^{-n - 1}$
  and $J_x^-(z) = -\sum_{n \ge 0} x[n] z^{-n - 1}$.
\end{definition}

\begin{remark}
  Note that $J_x^-(z)$ is a well-defined operator
  on a highest weight module $V$. Similarly, the
  products $J_x^+(z) J_y^+(\zeta)$,
  $J_x^-(z) J_y^-(\zeta)$, and
  $J_x^+(z) J_y^-(\zeta)$ are well-defined as
  operators on $\widehat{V}$. On the other hand,
  $J_x^-(z) J_y^+(\zeta)$ is not always well-defined:
  The components of $J_x^-(z) J_y^+(\zeta) u$ are of
  the form
  \[
    -\sum_{j \ge j_0} z^{-j - 1} w^{-n + j - 1}
    x[j] y[n - j] u. \tag{$**$}
  \]
  Nevertheless, $x[j] u = 0$ for $j$ sufficiently
  large, and so
  \[
    x[j] y[n - j] u
    = [x[j], y[n - j]] u
    = ([x, y])[n] u + j \langle x, y \rangle \delta_{n, 0} k u.
  \]
  Thus, analytically, the convergence of the series
  $(**)$ amounts to the convergence of
  \[
    \sum_{j \ge j_0} z^{-j - 1} \zeta^{-n + j - 1}
    \quad \text{and} \quad
    \sum_{j \ge j_0} j z^{-j - 1} \zeta^{-n + j - 1}.
  \]
  These converge for $|z| > |\zeta|$, so the
  operator $J_x^-(z) J_y^+(\zeta)$ is well-defined
  on $\widehat{V}$ in this region.
\end{remark}

\begin{theorem}
  We have the following:
  \begin{gather*}
    [J^{\pm}_x(z), J_y^{\pm}(w)]
    = \frac{1}{z - w} (J^{\pm}_{[x, y]}(z) - J^{\pm}_{[x, y]}(w)), \quad z \ne w; \\
    [J_x^-(z), J_y^+(w)]
    = \frac{1}{z - w} (J^-_{[x, y]}(z) - J^+_{[x, y]}(w))
    - \frac{k \langle x, y \rangle}{(z - w)^2},
    \quad |z| > |w|.
  \end{gather*}
\end{theorem}

\begin{corollary}
  We have the following:
  \[
    [J_x^{\pm}(z), J^{\pm}_y(z)]
    = \frac{d}{dz} J^{\pm}_{[x, y]}(z).
  \]
\end{corollary}

\begin{remark}
  Let $L(z) = \sum_{n \in \Z} L_n z^{-n - 2}$.
  Then the formulas $(*)$ from Section
  \ref{sec:virasoro} can be rewritten as
  \[
    L(z) = \frac{1}{2(k + h^\vee)} \sum_{a \in B}
    {:} J_a(z)^2 {:},
  \]
  where the normal ordered product is extended
  $\C[z, z^{-1}]$-linearly.
\end{remark}
