\chapter{Apr.~1 --- Vertex Operators}

\section{KZ Equations in Higher Rank}

\begin{remark}
  Let $\g$ be a simple Lie algebra.
  In the general setting, we are looking
  for $W = (V^{\mathfrak{n}_-})^\lambda$,
  where
  \[
    V = V_{\mu_1} \otimes \cdots \otimes V_{\mu_N},
  \]
  $\lambda = - \sum_{i = 1}^N \mu_i + \mu$
  for $\mu \in \Q^+$. We call
  $\mu = \sum \ell_i \alpha_i$ the
  \emph{multilevel} and $m = |\mu| = \sum \ell_i$
  the \emph{level} of the solution.
  Let $p_s = \sum_{j = 1}^s \ell_j$ and
  $\nu : \{1, \dots, m\} \to \{1, \dots r\}$
  (where $r$ is the rank of $\g$)
  such that $\nu(p) = s$ if
  $p_{s - 1} < p \le p_s$. Then we can define
  the function
  \[
    \psi_{\mu}(\vec{z}, \vec{t})
    = \sum_{1 \le i < j \le N} (z_i - z_j)^{\langle \mu_i, \mu_j \rangle / \kappa}
    \prod_{j = 1}^N \prod_{p = 1}^m (t_p - z_j)^{- \langle \alpha_{\nu(p)}, \mu_j \rangle / \kappa}
    \prod_{1 \le p < n \le m} (t_p - t_n)^{\langle \alpha_{\nu(p)}, \alpha_{\nu(n)} \rangle / \kappa}.
  \]
  This is the multi-valued part, which we can
  combine with a rational part to give solutions
  to the KZ equations.
  The details can be found in the paper
  by V. Schechtman and A. Varchenko.
\end{remark}

\section{Vertex Operators}

\begin{remark}
  Let $\mathfrak{H} = \mathfrak{h} \otimes \C[t, t^{-1}] \oplus \C c \subseteq \widehat{\g}$
  be the extended Cartan subalgebra in
  $\widehat{g}$, which is
  called the \emph{Heisenberg algebra}. For
  any $h, g \in \mathfrak{h}$, we have the
  commutation relation
  \[
    [h[m], g[n]]
    = m \langle h, g \rangle \delta_{m, -n} c.
  \]
  Let $H_\lambda$
  (for $\lambda \in \mathfrak{h}^*$)
  be the \emph{Fock module}
  over $\mathfrak{H}$, which is generated by
  a highest weight vector $v_\lambda$ with
  \[
    h[n] v_\lambda = 0 \text{ for } n > 0, \quad
    h[0] v_\lambda = \lambda(h) v_\lambda, \quad
    c v_\lambda = v_\lambda.
  \]
  We can decompose
  $\mathfrak{H} = \mathfrak{H}_+ \oplus \mathfrak{H}_0 \oplus \mathfrak{H}_-$,
  and $\mathfrak{H}_- = \mathfrak{h} \oplus t^{-1} \C[t^{-1}]$
  acts freely on $H_\lambda$. Write
  \[
    \mathfrak{H}_{\ne 0}
    = \mathfrak{H}_+ \oplus \mathfrak{H}_-
    \oplus \C c.
  \]
  Note that $H_\lambda$ does not
  notice $\lambda$, and we have
  \begin{align*}
    e^\mu
    : H_\lambda
    &\longrightarrow H_{\lambda + \mu} \\
    v_\lambda &\longmapsto v_{\lambda + \mu},
  \end{align*}
  which is an isomorphism of
  $\mathfrak{H}_{\ne 0}$-modules.
  Note that $e^{\lambda} e^\mu = e^{\lambda + \mu}$.

  Define the \emph{vertex operator}
  (for $z \in \C^\times, \mu \in \mathfrak{h}^*$)
  \[
    X(\mu, z)
    = \exp\bigg(
      \sum_{n < 0} \frac{h_\mu[n]}{-n} z^{-n}
    \bigg)
    \exp\bigg(
      \sum_{n > 0} \frac{h_\mu[n]}{-n} z^{-n}
    \bigg)
    e^\mu z^{h_\mu[0]},
  \]
  which acts via
  $H_\lambda \to \widehat{H}_{\lambda + \mu}$.
\end{remark}

\begin{definition}
  The \emph{normal ordered product}
  ${:}X(\mu_1, z_1) X(\mu_2, z_2){:}$ is defined so
  that
  \[
    {:}h_1[n] h_2[m]{:}
    =
    \begin{cases}
      h_1[n] h_2[m], & \text{if } m \ge 0, \\
      h_2[m] h_1[n], & \text{if } m < 0,
    \end{cases}
    \quad \text{and} \quad
    {:}h[0] e^\mu{:} = {:}e^\mu h[0] {:}
    = e^\mu h[0].
  \]
\end{definition}

\begin{prop}
  If $|z_1| > |z_2|$, then
  $X(\mu_1, z_1) X(\mu_2, z_2)$ is a
  well-defined operator
  $H_\lambda \to \widehat{H}_{\lambda + \mu_1 + \mu_2}$, and
  we have
  \[
    X(\mu_1, z_1) X(\mu_2, z_2)
    = (z_1 - z_2)^{(\mu_1, \mu_2)}
    {:}X(\mu_1, z_1) X(\mu_2, z_2){:}.
  \]
\end{prop}

\begin{proof}
  Note that by the Baker-Campbell-Hausdorff formula, we have
  $e^{A} e^{B} = e^B e^A e^{[A, B]}$
  in this case. Let
  \[
    M_2^-
    = \sum_{n < 0} \frac{h_{\mu_2}[n]}{-n} z_2^{-n}
    \quad \text{and} \quad
    M_1^+
    = -\sum_{n > 0} \frac{h_{\mu_1}[n]}{-n} z_1^{-n}
  \]
  Then we have
  \begin{align*}
    [M_1^+, M_2^-]
    &= -\sum_{m, n > 0} \frac{[h_{\mu_1}[n], h_{\mu_2}[-m]]}{m n} z_1^{-n} z_2^m
    = - \sum_{n > 0} \frac{n \langle h_{\mu_1}, h_{\mu_2} \rangle}{n^2} \bigg(\frac{z_2}{z_1}\bigg)^n \\
    &= - \sum_{n > 0} \frac{\langle \mu_1, \mu_2 \rangle}{n} \bigg(\frac{z_2}{z_1}\bigg)^n
    = \langle \mu_1, \mu_2 \rangle \log\bigg(1 - \frac{z_2}{z_1}\bigg),
  \end{align*}
  so we see that
  $e^{M_1^+} e^{M_2^-} = e^{M_2^-} e^{M_1^+} (1 - z_2 / z_1)^{\langle \mu_1, \mu_2 \rangle}$.
\end{proof}

\begin{corollary}
  We have
  \[
    X(\mu_1, z_1) \cdots X(\mu_n, z_n)
    =
    \prod_{i < j} (z_i - z_j)^{(\mu_i, \mu_j)}
    {:}X(\mu_1, z_1) \cdots X(\mu_n, z_n){:}.
  \]
\end{corollary}

\begin{prop}
  We have the following:
  \begin{enumerate}
    \item $[h_\lambda[n], X(\mu, z)] = \langle \lambda, \mu \rangle z^n X(\mu, z)$;
    \item for $h(z) = \sum h[n] z^{-n - 1}$
      and $|z_1| > |z_2|$,
      we have
      \[
        h_\lambda(z_1) X(\mu, z_2)
        = {:}h_\lambda(z_1) X(\mu, z_2){:}
        + \frac{(\lambda, \mu)}{z_1 - z_2} X(\mu, z_2)
        \quad \text{and} \quad
        \frac{d}{dz} X(\mu, z) {:} h_\mu(z) X(\mu, z) {:}.
      \]
  \end{enumerate}
\end{prop}

\begin{remark}
  Let $g$ be a Lie algebra of
  type ADE. The \emph{basic representation}
  of $\widehat{g}$ is
  $L_{0, 1}$, and
  \[L_{0, 1} = \bigoplus_{\beta \in Q} H_\beta\]
  by the
  Weyl-Kac character formula,
  where $Q$ is the root lattice of $g$.
  Note that $\widehat{g}$ can be
  represented by $h[n]$ and $e^\mu$ (i.e.
  the action of $\widehat{g}$ is generated by
  $h[n]$ and $e^\mu$).
\end{remark}

\begin{theorem}
  Let $\g$ be a Lie algebra of type ADE, so
  that $\langle \alpha, \alpha \rangle = 2$
  for all $\alpha \in R$. Let
  \[
    \varepsilon : Q \times Q \longrightarrow \{\pm 1\}
  \]
  such that $\varepsilon(\alpha, \beta) \varepsilon(\beta, \alpha) = (-1)^{(\alpha, \beta)}$,
  where $Q$ is the root lattice of $\g$.
  Then $L_{0, 1} = \bigoplus_{\beta \in Q} H_\beta$
  such that the action of $\widehat{g}$
  on $L_{0, 1}$ is given by
  \[
    J_{h}(z) \sum_{n \in \Z} h[n] z^{-n - 1}, \quad
    J_{e_\alpha}(z)
    = X(\alpha, z) C_\alpha, \quad
    J_{f_\alpha}(z)
    = X(-\alpha, z) C_{-\alpha},
  \]
  where $J_x = \sum x[n] z^{-n - 1}$ for
  $x \in \g$ and $C_\alpha|_{H_\beta} = \varepsilon(\alpha, \beta)$.
\end{theorem}

\begin{definition}
  Define $\mu' = \mu / \sqrt{\kappa}$
  (where $\kappa = k + h^\vee$), and
  define a linear functional
  $v^*_{\mu'}$ on $H_{\mu'}$ by
  $(v^*_{\mu'}, v_{\mu'}) = 1$ and
  $(v^*_{\mu'}, v) = 0$ if $v$ is a
  homogeneous vector of nonzero degree in
  $H_{\mu'}$.
\end{definition}

\begin{prop}
  We have
  \[
    \langle v^*_{\mu'}, X(\mu_1', z_1) \cdots X(\mu_N', z_N) v_{\mu'} \rangle
    = \prod_{i < j} (z_i - z_j)^{\langle \mu_i, \mu_j \rangle / \kappa}
    = \psi_0(z_1, \dots, z_N),
  \]
  where $\mu = \sum_{i = 1}^n \mu_i$.
\end{prop}

\begin{corollary}
  We have
  \[
    \psi_\mu(\vec{z}, \vec{t})
    = \langle v^*_{-\lambda'}, X(-\alpha_{\nu(1)}', t_1) \cdots X(-\alpha_{\nu(m)}', t_m) X(\mu_1', z_1) \cdots X(\mu_N', z_N) v_{0} \rangle,
  \]
  where $\lambda = -\sum_{i = 1}^N \mu_i + \mu$.
\end{corollary}

\section{The \texorpdfstring{$\beta$-$\gamma$}{beta-gamma} System}

\begin{remark}
  Consider the \emph{$\beta$-$\gamma$ system}
  $B$, which is
  generated by $\beta_n, \gamma_n, c$ for
  $n \in \Z$ with
  relations
  \[
    [\beta_n, \gamma_m] = \delta_{n, -m} c,
    \quad
    [\beta_n, \beta_m] = [\gamma_n, \gamma_m] = 0, \quad
    [c, \beta_n] = [c, \gamma_n] = 0.
  \]
  Define $\beta(z) = \sum_{n \in \Z} \beta_n z^{-n - 1}$
  and $\gamma(z) = \sum_{n \in \Z} \gamma_n z^{-n}$, then
  \[
    {:} \beta_n \gamma_m {:}
    = {:} \gamma_m \beta_n {:}
    =
    \begin{cases}
      \beta_n \gamma_m, & \text{if } n < 0, \\
      \gamma_m \beta_n, & \text{if } n \ge 0.
    \end{cases}
  \]
  The \emph{Fock module} $H(B)$ over $B$ is
  generated by $v$ with
  $cv = v$, $\gamma_n v = 0$ for $n > 0$,
  and $\beta_n v = 0$ for $n \ge 0$.
\end{remark}
