\section{The \texorpdfstring{$\beta$-$\gamma$}{beta-gamma} System, Continued}

\begin{remark}
  Let $z \in \C^\times$, then
  $\beta(z), \gamma(z)$ are well-defined
  operators $H(B) \to \widehat{H}(B)$.
  Moreover,
  ${:} \beta(z_1) \gamma(z_2) {:}$
  is also well-defined as an operator
  $H(B) \to \widehat{H}(B)$.
\end{remark}

\begin{prop}
  If $|z_1| > |z_2|$, then
  \[
    \beta(z_1) \gamma(z_2)
    = {:} \beta(z_1) \gamma(z_2) {:}
     + \frac{c}{z_1 - z_2}.
  \]
\end{prop}

\begin{proof}
  We have
  \[
    \beta(z_1) \gamma(z_2)
    - {:} \beta(z_1) \gamma(z_2) {:}
    = \sum_{n \ge 0} [\beta_n, \gamma_{-n}]
    z_1^{-n - 1} z_2^n
    = c \sum_{n \ge 0}  z_1^{-n - 1} z_2^n
    = \frac{c}{z_1 - z_2},
  \]
  which is the desired result.
\end{proof}

\begin{remark}
  Let $H$ be a vector space and
  $\omega$ and antisymmetric bilinear
  form on $H$. Let
  $\widehat{H} = H \oplus \C c$ with
  \[
    [h_1 + \lambda_1 c, h_2 + \lambda_2 c]
    = \omega(h_1, h_2) c.
  \]
  Then we have
  $\widehat{H} = H^+ \oplus H^0 \oplus H^- \oplus \C c$,
  where $H^+, H^-$ are isotropic subspaces
  with respect to $\omega$ and
  $H^0 = \ker \omega$. We can define
  relations
  \[
    cu = u, \quad
    h^+ u = h^0 u = 0 \text{ for }
    h^+ \in H^+,\, h^0 \in H^0.
  \]
  We call $u$ the \emph{vacuum vector}.
  Note that $\mathcal{F} = \C u \oplus H^- \mathcal{F}$,
  and if $u^*$ is the linear functional
  with $\langle u^*, u \rangle = 1$, then
  $u^*$ vanishes on $H^- \mathcal{F}$.
\end{remark}

\begin{theorem}[Wick]
  Denote by $P(N)$ the set of partitions
  of $\{1, \dots, N\}$ into $N / 2$ unordered
  pairs of numbers (with $P(N) = \varnothing$ for
  odd $n$).
  Let $h_1, \dots, h_N \in H$. Then
  \[
    \langle u^*, h_1 \cdots h_N u \rangle
    = \sum_{p \in P(N)}
    \prod_{n = 1}^{N / 2}
    \langle u^*, h_{p_1(n)} h_{p_2(n)} u \rangle
    = \sum_{p \in P(N)}
    \prod_{n = 1}^{N / 2}
    \omega(h_{p_1(n)}, h_{p_2(n)}),
  \]
  where $p_1(n) < p_2(n)$ is the
  $n$th pair in $p$.
\end{theorem}

\begin{remark}
  Consider a correlation function of the
  type
  \[
    \rho_{\vec{m}}(\vec{z}, \vec{t})
    = \frac{1}{m_1! \cdots m_N!}
    \langle u^*, \beta(t_1) \cdots \beta(t_m)
    \gamma(z_1)^{m_1} \cdots \gamma(z_N)^{m_N} u \rangle.
  \]
  \pagebreak
  By Wick's theorem, this reduces to
  \[
    \rho_{\vec{m}}(\vec{z}, \vec{t})
    = m_1! \cdots m_N!
    \sum_{s \in \Sigma_{\vec{m}}}
    \prod_{n = 1}^m \langle u^*, \beta(t_n) \gamma(z_{s(n)}) u \rangle,
  \]
  where for $z \in \C^\times$ we have
  \[
    \langle u^*, \beta(t) \gamma(z) u \rangle
    = \frac{1}{t - z} +
    \langle u^*, {:} \beta(t) \gamma(z) {:} u \rangle
    = \frac{1}{t - z}.
  \]
\end{remark}

\section{Factorization of KZ Solutions for \texorpdfstring{$\mathfrak{sl}_2$}{sl2}}

\begin{remark}
  Consider the case for $\g = \mathfrak{sl}_2$.
  Let $U(t) = X(-\alpha', t)$, where
  $\alpha$ is the positive root of
  $\mathfrak{sl}_2$, and
  $Z_m(\lambda, z) = X(\lambda', z)
    \otimes \gamma^m(z)$.
  Then we have actions
  \[
    \begin{tikzcd}
      & H_\mu \otimes H(B) \ar[dl, "U(t)", swap] \ar[dr, "{Z_m(\lambda, z)}"] \\
      \widehat{H}_{\mu - \alpha'} \otimes \widehat{H}(B)
      & & \widehat{H}_{\mu + \lambda'} \otimes \widehat{H}(B)
    \end{tikzcd}
  \]
  Then we can write solutions to the KZ
  equations as
  \begin{align*}
    \psi_C(\vec{z})
    &= \sum_{\vec{m}} \int_C \langle
    v_{\lambda_0} \otimes u^*, U(t_1) \cdots U(t_m) Z_{m_1}(\lambda_1, z_1)
    \cdots Z_{m_N}(\lambda_N, z_N) v_0 \otimes u \rangle\, d\vec{t} \\
    & \quad \quad \quad \quad \quad {e^{m_1}}{m_1!} v_1 \otimes \cdots \otimes \frac{e^{m_N}}{m_N!} v_N,
  \end{align*}
  where $\lambda_0 = \sum \mu_i - 2m$.
\end{remark}

\section{Intertwining Operators via Fock Modules}

\begin{remark}
  Let $\alpha$ be the positive root for
  $\mathfrak{sl}_2$.
  Define an algebra $A$ by 
  \[
    [\alpha_n, \alpha_m] = 2n \delta_{n, -m} c
    \quad \text{and} \quad
    [c, \alpha_n] = 0,
  \]
  where $\alpha_n = h_\alpha[n]$, and define
  a module
  $H_\lambda(A)$ generated by $v$ with
  \[
    cv = v, \quad \alpha_n v = 0 \text{ for } n > 0, \quad
    \alpha_0 v = \lambda v
    \text{ for } \lambda \in \C.
  \]
  Let
  $\mathfrak{h}^* \cong \C$ where
  $\alpha \mapsto 2$, so that
  \[
    X(\mu, z) = X(\mu \alpha / 2, z),
    \quad \mu \in \C.
  \]
  Let $\alpha(z) = \sum_{n \in \Z} \alpha_n z^{-n - 1}$
  (compute $\alpha(z_1) \alpha(z_2)$ as
  an exercise for $|z_1| > |z_2|$). Then
  \[
    J_e(z) = \beta(z), \quad
    J_h(z) = -2 {:} \gamma(z) \beta(z) {:} + \sqrt{\kappa} \alpha(z), \quad
    J_f(z) = - {:} \gamma(z)^2 \beta(z) {:} + \sqrt{\kappa} \alpha(z) \gamma(z) + k \frac{d\gamma(z)}{dz}
  \]
  gives an action of the affine Lie
  algebra
  $\widehat{\mathfrak{sl}_2}$
  on $H_\lambda(A) \otimes H(B)$.
\end{remark}

\begin{remark}
  Let $\beta = d / dx$, $\gamma = x$, then
  \[
    e = \beta, \quad h = -2 \gamma \beta + j,
    \quad f = -\gamma^2 \beta + j \gamma
  \]
  define a representation of
  the contragradient Verma module of highest
  weight $j$ on $\C[z]$. There is then a
  notion of ``affinization'' which
  yields the above action for
  $\widehat{\mathfrak{sl}_2}$.
\end{remark}

\begin{theorem}[Wakimoto realization]
  Define an $\widehat{\mathfrak{sl}_2}$-action
  on $H_\lambda(A) \otimes H(B)$ as above.
  If $\lambda, k$ are such that
  $M_{\sqrt{\kappa} \lambda, k}$ is irreducible,
  then this representation is isomorphic
  to $M_{\sqrt{\kappa} \lambda, k}$.
\end{theorem}

\begin{remark}
  Let $\g = \mathfrak{sl}_2$ and
  $\lambda, \lambda + \mu, k$
  be generic. Then for all $m \ge 0$, there
  is a unique (up to scalar)
  intertwiner of level $m$
  \[
    \Phi^m(z)
    : M_{\lambda, k}
    \longrightarrow \widehat{M}_{\lambda + \mu - 2m, k} \otimes V_\mu(z),
  \]
  where $V_\mu$ is the lowest weight Verma
  module. Writing
  $\mathcal{H}_{\lambda'} = H_\lambda(A) \otimes H(B)$, we have
  \[
    \Phi^m(z)
    : \mathcal{H}_{\lambda'}
    \longrightarrow
    \mathcal{H}_{\lambda' + \mu' - 2m'}
    \otimes V_\mu(z).
  \]
\end{remark}

\begin{lemma}
  We have
  \[
    \Phi^0 w
    = X(\mu', z) e^{-\gamma(z) \otimes e}
    (w \otimes v_\mu),
  \]
  where $w \in \mathcal{H}_{\lambda'}$
  and $v_\mu$ is the lowest weight vector
  of $V_\mu(z)$.
\end{lemma}

\begin{proof}
  Let $E(z) = \exp(-\gamma(z) \otimes e)$. Then
  one can compute that
  \[
    [X(\mu', z) E(z), J_x^n \otimes 1]
    w \otimes v_\mu
    = z^n(1 \otimes x) X(\mu', z) E(z)
    w \otimes V_\mu
  \]
  for all $x \in \mathfrak{sl}_2$ and $w \in \mathcal{H}_{\lambda'}$.
\end{proof}

\begin{lemma}
  Let $U(t) = X(-\alpha', t) \beta(t)$,
  where $\alpha' = \alpha / \sqrt{\kappa}$.
  Then
  \[
    [J^n_e, U(t)] = 0 \quad \text{and} \quad
    [J_f^n, U(t)]
    = \kappa \frac{d}{dt}(t^n X(-\alpha', t)).
  \]
\end{lemma}

\begin{remark}
  Consider the expression
  \[
    \int_C U(t)\, dt
    : \mathcal{H}_{\lambda'}
    \longrightarrow
    \widehat{\mathcal{H}}_{\lambda' - \alpha'}.
  \]
  For generic $\lambda$, such an operator
  does not exist.
\end{remark}

\begin{remark}
  Define an operator
  \begin{align*}
    \Xi : \mathcal{H}_{\lambda'}
    &\longrightarrow
    \widehat{\mathcal{H}}_{\lambda' + \mu' - 2m'}
    \otimes V_\mu \\
    (z, t_1, \dots, t_m)
    &\longmapsto
    X(\mu', z) e^{-\gamma(z) \otimes e}
    U(t_1) \cdots U(t_m) (w \otimes v_\mu),
  \end{align*}
  valid for
  $|z| > |t_1| > \cdots > |t_m| > 0$
  (but can be extended to a larger region
  via analytic continuation).
\end{remark}

\begin{prop}
  We have
  \[
    \Xi(z, t_1, \dots, t_m)
    =
    z^{\lambda \mu / \kappa}
    \prod_{j = 1}^m
    (z - t_j)^{1 - \mu / \kappa}
    t_j^{-\lambda / \kappa}
    \prod_{i < j} (t_i - t_j)^{2 / \kappa}
    \Xi_0(z, t_1, \dots, t_m),
  \]
  where $\Xi_0$ is a meromorphic operator-valued
  function on
  $(\C^\times)^{m + 1}$.
\end{prop}

\begin{proof}
  We have
  \[
    e^{-\gamma(z) \otimes e}
    \beta(t)
    = - \frac{1 \otimes e}{t - z} e^{-\gamma(z) \otimes e}
    + {:} e^{-\gamma(z) \otimes e} \beta(t) {:}
    = {:} \exp(-\gamma(z) \otimes e)
    \bigg(
      \beta(t) - \frac{1 \otimes e}{t - z}
    \bigg){:},
  \]
  and thus we see that
  \[
    e^{-\gamma(z) \otimes e}
    \beta(t_1) \cdots \beta(t_m)
    = {:}\exp(-\gamma(z) \otimes e)
    \bigg(
      \beta(t_1) - \frac{1 \otimes e}{t_1 - z}
    \bigg) \cdots
    \bigg(
      \beta(t_m) - \frac{1 \otimes e}{t_m - z}
    \bigg){:},
  \]
  which implies the result.
\end{proof}
